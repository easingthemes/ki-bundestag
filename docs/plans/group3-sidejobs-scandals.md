# Group 3: Side Jobs & Scandals

> Detailed implementation plan
> Parent: [docs/plans/abgeordnetenwatch-feature-roadmap.md](./abgeordnetenwatch-feature-roadmap.md)
> Status: Not started

## Overview

Add a side job (Nebentätigkeiten) system for AI-controlled MdBs. Side jobs are generated by the simulation engine, inspired by real-world data from abgeordnetenwatch. Controversial side jobs trigger media scandals that affect party approval. Displayed on MdB profiles and party detail pages.

**Design constraint**: Side jobs are simulation-internal fiction. Real-world sidejob data from abgeordnetenwatch is used ONLY as creative inspiration for the AI generator (organization types, income levels, categories). Never present real politician names or real side jobs as simulation facts.

## Step 2: Sidejob generation engine

**File**: `packages/engine/src/simulation/sidejobs.ts` (NEW)

### 2.1 Generation cycle

- Run every ~30 sim days (check `currentDay % 30 === 0`)
- Generate 2-5 new sidejobs per cycle for random AI-controlled MdBs
- ~20% chance a sidejob is controversial (high income + lobbying-adjacent category)

### 2.2 Batch AI generation

```typescript
export function shouldGenerateSidejobs(currentDay: number): boolean {
  return currentDay > 0 && currentDay % 30 === 0;
}

export function buildSidejobBatchRequest(
  currentDay: number,
  parties: Party[],
  activeAiSeats: BundestagSeat[],
): BatchRequest {
  // Pick 3-5 random AI seats
  const candidates = activeAiSeats
    .sort(() => Math.random() - 0.5)
    .slice(0, 3 + Math.floor(Math.random() * 3));

  const seatList = candidates.map(s => {
    const party = parties.find(p => p.id === s.partyId);
    return `Sitz ${s.seatNumber} (${party?.name ?? s.partyId})`;
  }).join("\n");

  return {
    customId: "sidejob-generation",
    system: SIDEJOB_SYSTEM_PROMPT,
    prompt: `Generiere realistische Nebentätigkeiten für diese MdBs:\n${seatList}\n\nAktueller Simulationstag: ${currentDay}`,
    maxTokens: 1024,
    roleKey: "daily",
  };
}
```

### 2.3 System prompt

const SIDEJOB_SYSTEM_PROMPT = `Du bist ein Parlamentssimulator. Generiere realistische Nebentätigkeiten für Bundestagsabgeordnete.

Einkommensstufen: "1000-3500", "3500-7000", "7000-15000", "15000-30000", "30000+"

### 2.4 Result processing

```typescript
export function processSidejobResult(
  result: BatchResult | undefined,
  candidates: BundestagSeat[],
  currentDay: number,
): SimulationEvent[] {
  // Parse JSON, validate, insert into sidejobs table
  // Return events for controversial sidejobs (for media generation)
  const events: SimulationEvent[] = [];

  for (const job of parsed.sidejobs) {
    const seat = candidates[job.seatIndex];
    if (!seat) continue;

    db.insert(schema.sidejobs).values({
      id: genId(),
      seatId: seat.id,
      partyId: seat.partyId,
      politicianName: job.politicianName,
      organization: job.organization,
      role: job.role,
      incomeLevel: job.incomeLevel,
      category: job.category,
      isControversial: job.isControversial,
      createdOnDay: currentDay,
      active: true,
    }).run();

    if (job.isControversial) {
      events.push({
        dayNumber: currentDay,
        type: "sidejob_scandal",
        actor: seat.partyId,
        title: `Kontroverse Nebentätigkeit: ${job.politicianName} (${seat.partyId.toUpperCase()})`,
        description: `${job.politicianName} erhält ${job.incomeLevel}€ von ${job.organization} als ${job.role}`,
        data: { seatId: seat.id, partyId: seat.partyId, organization: job.organization },
      });
    }
  }

  return events;
}
```

### 3.2 Approval impact

**File**: `packages/engine/src/simulation/sidejobs.ts`

```typescript
export function applySidejobScandalImpact(
  events: SimulationEvent[],
  parties: Party[],
): void {
  const scandals = events.filter(e => e.type === "sidejob_scandal");
  for (const scandal of scandals) {
    const partyId = scandal.data?.partyId as string;
    const party = parties.find(p => p.id === partyId);
    if (party) {
      // -0.2 to -0.5 approval hit
      const impact = -(0.2 + Math.random() * 0.3);
      party.approvalRating = Math.max(5, party.approvalRating + impact);
    }
  }
}
```

  if (aiSeats.length > 0) {
    const sjReq = buildSidejobBatchRequest(currentDay, allParties, aiSeats);
    const sjResults = await submitBatch([sjReq]);

## Step 4: API endpoints

**File**: `packages/api/src/routes/parliament.ts`

### 4.1 `GET /api/parliament/sidejobs`

```typescript
// Query params: ?partyId=spd&controversial=true
// Response: Sidejob[]
router.get("/sidejobs", (req, res) => {
  let query = db.select().from(schema.sidejobs)
    .where(eq(schema.sidejobs.active, true));

  if (req.query.partyId) {
    query = query.where(eq(schema.sidejobs.partyId, req.query.partyId as string));
  }

  const sidejobs = query.orderBy(desc(schema.sidejobs.createdOnDay)).all();
  res.json(sidejobs);
});
```

### 4.2 `GET /api/parliament/sidejobs/seat/:seatId`

```typescript
// Response: Sidejob[] for a specific MdB
router.get("/sidejobs/seat/:seatId", (req, res) => {
  const sidejobs = db.select().from(schema.sidejobs)
    .where(and(
      eq(schema.sidejobs.seatId, req.params.seatId),
      eq(schema.sidejobs.active, true),
    ))
    .orderBy(desc(schema.sidejobs.createdOnDay))
    .all();
  res.json(sidejobs);
});
```
